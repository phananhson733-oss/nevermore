function Table({ cols, rows, minW }) {
  return (
    <div className="p-x" style={{ border: `1px solid ${T.line}` }}>
      <table style={{ width: "100%", borderCollapse: "collapse", minWidth: minW || 480 }}>
        <thead>
          <tr>{cols.map((h) => (
            <th key={h} className="p-mono" style={{ padding: "9px 13px", textAlign: "left", fontSize: 10, fontWeight: 400, color: T.text3, background: T.panel, borderBottom: `1px solid ${T.line}`, whiteSpace: "nowrap" }}>{h}</th>
          ))}</tr>
        </thead>
        <tbody>
          {rows.map((r, i) => (
            <tr key={i}>{r.map((cell, j) => {
              const arr = Array.isArray(cell);
              const txt = arr ? cell[0] : cell;
              const col = arr ? cell[1] : T.text2;
              const dashed = arr && cell[2] === "dashed";
              return (
                <td key={j} className="p-mono" style={{ padding: "9px 13px", fontSize: 11, borderBottom: `1px solid ${T.line}`, lineHeight: 1.7, color: col, whiteSpace: j === 0 ? "normal" : "nowrap" }}>
                  {dashed ? <span style={{ border: `1px dashed ${col}`, padding: "1px 5px" }}>{txt}</span> : txt}
                </td>
              );
            })}</tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}

const ST = { pass: [T.pass, "通过"], fail: [T.fail, "未通过"], na: [T.text3, "抓取失败"], skip: [T.text3, "不适用"] };

function Checks({ items }) {
  return (
    <div style={{ border: `1px solid ${T.line}` }}>
      {items.map((it, i) => {
        const [col, label] = ST[it[0]];
        const dashed = it[0] === "na" || it[0] === "skip";
        return (
          <div key={i} className="p-row" style={{ padding: "10px 14px", alignItems: "flex-start", borderTop: i === 0 ? "none" : `1px solid ${T.line}`, background: T.panel }}>
            <span className="p-mono" style={{ fontSize: 10, padding: "1px 6px", color: col, border: `1px ${dashed ? "dashed" : "solid"} ${col}`, whiteSpace: "nowrap" }}>{label}</span>
            <div style={{ flex: "1 1 220px", minWidth: 0 }}>
              <div style={{ fontSize: 12, color: T.text, lineHeight: 1.7 }}>
                {it[1]}
                {it[3] === "h" && <span className="p-mono" style={{ fontSize: 9, marginLeft: 7, padding: "0 4px", border: `1px solid ${T.brass}`, color: T.brass }}>启发式</span>}
              </div>
              <div className="p-mono" style={{ fontSize: 10, color: T.text3, marginTop: 3, lineHeight: 1.7 }}>{it[2]}</div>
              {it[4] && <div style={{ fontSize: 10.5, color: T.text2, marginTop: 5, lineHeight: 1.7 }}>修法：{it[4]}</div>}
            </div>
          </div>
        );
      })}
    </div>
  );
}

function Notice({ tone, title, body }) {
  const c = tone === "warn" ? T.brass : tone === "good" ? T.pass : tone === "bad" ? T.fail : "#6E97D6";
  return (
    <div style={{ padding: 15, background: T.panel, border: `1px solid ${T.line}`, borderLeft: `3px solid ${c}` }}>
      <div className="p-mono" style={{ fontSize: 11, color: c, marginBottom: 6 }}>{title}</div>
      <div style={{ fontSize: 12, lineHeight: 1.9, color: T.text2 }}>{body}</div>
    </div>
  );
}
